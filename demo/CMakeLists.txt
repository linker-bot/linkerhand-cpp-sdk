cmake_minimum_required(VERSION 3.16)
project(legacy_sdk_readonly LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
# Supports both the package tree and a separately installed SDK.
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/..")
find_package(linkerhand-legacy-cpp-sdk 2.2.1 CONFIG REQUIRED)
add_executable(demo main.cpp)
target_link_libraries(demo PRIVATE LinkerHand::linkerhand_legacy_cpp_sdk)
set_target_properties(demo PROPERTIES BUILD_RPATH "${linkerhand-legacy-cpp-sdk_LIBRARY_DIR}")
if(WIN32)
    add_custom_command(TARGET demo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:LinkerHand::linkerhand_legacy_cpp_sdk> $<TARGET_FILE_DIR:demo>)
endif()
